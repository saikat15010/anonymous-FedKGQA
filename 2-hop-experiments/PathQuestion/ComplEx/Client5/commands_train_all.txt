cd "/home/islamm9/PathQuestion_Dataset/ComplEx/Client5"


# ========================================
# BERT-ComplEx Training (Client 5)
# ========================================

python main_bert_complex.py \
    --data_path /home/islamm9/PathQuestion_Dataset/Dataset/Client5/federated_clients \
    --num_clients 5 \
    --hidden_dim 512 \
    --kg_max_rounds 50 \
    --qa_max_rounds 50 \
    --local_epoch 3 \
    --qa_local_epoch 2 \
    --batch_size 512 \
    --qa_batch_size 8 \
    --qa_lr 1e-5 \
    --num_neg 256 \
    --num_neg_qa 128 \
    --check_per_round 5 \
    --state_dir ./state_bert_client5 \
    --log_dir ./log_bert_client5 \
    --name bert_complex_client5 \
    --gpu 0
