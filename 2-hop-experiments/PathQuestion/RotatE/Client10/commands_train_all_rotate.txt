    --data_path /home/islamm9/PathQuestion_Dataset/Dataset/Client10/federated_clients \
    --num_clients 10 \
    --hidden_dim 512 \
    --kg_max_rounds 50 \
    --qa_max_rounds 50 \

    --qa_local_epoch 2 \
    --batch_size 512 \

    --state_dir ./state_bert_client10 \

# ========================================
# NOTES:
# ========================================
# - KG rounds: 50 (trains RotatE embeddings)
# - QA rounds: 50 (trains language model QA head)
# - Evaluation every 5 rounds 
# - Best model saved based on Hits@5 on dev set
# - Training takes ~10-30 minutes per model depending on GPU
# - Models saved to ./state_*/best_models/
# - Logs saved to ./log_*/
# 
# RotatE Model:
# - Models relations as rotations in complex space
# - entities: 2*hidden_dim (complex embeddings)
# - relations: hidden_dim (phase angles)
# - Scoring: -||h ∘ r - t|| (rotation + distance)
# - Good for modeling symmetric AND asymmetric relations
# - Often achieves best accuracy among distance-based models
