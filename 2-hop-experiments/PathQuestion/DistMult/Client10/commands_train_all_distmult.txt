cd "/home/islamm9/PathQuestion_Dataset/DistMult/Client10"


# ========================================
# BERT-DistMult Training (Client 10)
# ========================================

python main_bert_distmult.py \
    --data_path /home/islamm9/PathQuestion_Dataset/Dataset/Client10/federated_clients \
    --num_clients 10 \
    --hidden_dim 512 \
    --kg_max_rounds 50 \
    --qa_max_rounds 50 \
    --local_epoch 3 \
    --qa_local_epoch 2 \
    --batch_size 512 \
    --qa_batch_size 8 \
    --qa_lr 1e-5 \
    --num_neg 256 \
    --num_neg_qa 128 \
    --check_per_round 5 \
    --state_dir ./state_bert_client10 \
    --log_dir ./log_bert_client10 \
    --name bert_distmult_client10 \
    --gpu 0

# ========================================
# NOTES:
# ========================================
# - KG rounds: 50 (trains DistMult embeddings)
# - QA rounds: 50 (trains language model QA head)
# - Evaluation every 5 rounds 
# - Best model saved based on Hits@5 on dev set
# - Training takes ~10-30 minutes per model depending on GPU
# - Models saved to ./state_*/best_models/
# - Logs saved to ./log_*/
# 
# DistMult Model:
# - Simplest KGE model (bilinear scoring)
# - entities: hidden_dim, relations: hidden_dim
# - Scoring: sum(h ⊙ r ⊙ t) element-wise product
# - Symmetric relations (cannot model inverse relations well)
# - Fastest training, lowest memory
